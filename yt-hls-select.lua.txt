-- yt-hls-select.lua
-- MPV 0.33 - Windows 7 x64
--
-- Fonctionnement :
--   1. Télécharge le manifeste maître HLS avec PowerShell.
--   2. Choisit la meilleure vidéo <= 720p, H.264 prioritaire.
--   3. Ouvre directement l'URL de la variante vidéo.
--   4. Ajoute séparément la piste audio "original".
--   5. Peut ajouter les autres langues avec KEEP_ALL_AUDIO_LANGUAGES.
--
-- Important :
--   Aucun manifeste maître temporaire n'est créé.
--   Aucun réglage de protocol_whitelist n'est nécessaire.

local utils = require 'mp.utils'

local TARGET_H = 720
local PREFER_H264 = true

-- false : ne charge que l'audio original (démarrage plus léger).
-- true  : ajoute aussi les autres langues du groupe audio choisi.
local KEEP_ALL_AUDIO_LANGUAGES = false

local MAX_MANIFEST_SIZE = 4 * 1024 * 1024

local pending_audio = nil
local audio_add_started = false

-- Catalogue des langues trouvées dans le manifeste.
-- Les pistes ne sont chargées que lorsqu'elles sont sélectionnées.
local audio_catalog = {}
local loaded_audio_ids = {}

local current_audio_index = 1
local current_audio_track_id = nil
local pending_audio_load = nil


local function lower(s)
    return tostring(s or ''):lower()
end


local function trim(s)
    return tostring(s or ''):gsub('^%s+', ''):gsub('%s+$', '')
end


local function contains_original(s)
    return lower(s):find('original', 1, true) ~= nil
end


-- Sépare les attributs HLS sans couper les virgules
-- qui se trouvent à l'intérieur des guillemets.
local function split_attribute_tokens(s)
    local tokens = {}
    local field = {}
    local in_quotes = false

    for i = 1, #s do
        local ch = s:sub(i, i)

        if ch == '"' then
            in_quotes = not in_quotes
            field[#field + 1] = ch

        elseif ch == ',' and not in_quotes then
            tokens[#tokens + 1] = table.concat(field)
            field = {}

        else
            field[#field + 1] = ch
        end
    end

    if #field > 0 then
        tokens[#tokens + 1] = table.concat(field)
    end

    return tokens
end


local function parse_hls_attributes(line)
    local attrs = {}
    local _, colon = line:find('^[^:]+:')

    if not colon then
        return attrs
    end

    local raw = line:sub(colon + 1)

    for _, token in ipairs(split_attribute_tokens(raw)) do
        local eq = token:find('=', 1, true)

        if eq then
            local key = trim(token:sub(1, eq - 1)):upper()
            local value = trim(token:sub(eq + 1))

            if #value >= 2
                and value:sub(1, 1) == '"'
                and value:sub(-1) == '"' then

                value = value:sub(2, -2)
            end

            attrs[key] = value
        end
    end

    return attrs
end

local function find_curl_exe()
    local path = os.getenv('PATH') or ''

    for dir in path:gmatch('[^;]+') do
        dir = trim(dir)
        dir = dir:gsub('^"(.*)"$', '%1')
        dir = dir:gsub('[\\/]+$', '')

        if dir ~= '' then
            local candidate = dir .. '\\curl.exe'

            if utils.file_info(candidate) then
                return candidate
            end
        end
    end

    return nil
end


local function fetch_manifest(url)
    if type(url) ~= 'string'
        or not url:match('^https?://') then
        return nil, 'URL HTTP(S) invalide'
    end
	
	    local curl_path = find_curl_exe()

    if curl_path then
        local start_time = mp.get_time()

        local ok, result = pcall(utils.subprocess, {
            args = {
                curl_path,
				'--insecure',
                '--disable',
                '--location',
                '--fail',
                '--silent',
                '--show-error',
                '--compressed',
                '--connect-timeout', '4',
                '--max-time', '6',
                '--user-agent', 'Mozilla/5.0',
                url
            },
            capture_stdout = true,
            capture_stderr = true,
            max_size = MAX_MANIFEST_SIZE,
            playback_only = true
        })

        local elapsed = mp.get_time() - start_time

        if ok
            and result
            and result.status == 0
            and type(result.stdout) == 'string'
            and result.stdout:find('#EXTM3U', 1, true) then

            mp.msg.info(string.format(
                'yt-hls-select : méthode=curl ; téléchargement total=%.2f s',
                elapsed))

            return result.stdout
        end

        local detail = result and
            (result.stderr or result.error or result.error_string)
            or tostring(result or 'erreur inconnue')

        mp.msg.warn(string.format(
            'yt-hls-select : curl a échoué après %.2f s ; essai avec PowerShell : %s',
            elapsed,
            tostring(detail)))
    else
        mp.msg.info(
            'yt-hls-select : curl.exe absent du PATH ; utilisation de PowerShell')
    end

    local quoted_url = "'" .. url:gsub("'", "''") .. "'"

    local ps =
        "$ErrorActionPreference='Stop'; " ..

        "try { " ..
        "[Net.ServicePointManager]::SecurityProtocol = " ..
        "[Net.SecurityProtocolType]::Tls12 " ..
        "} catch {}; " ..

        "$total = [Diagnostics.Stopwatch]::StartNew(); " ..

        "$req = [System.Net.HttpWebRequest]::Create(" ..
        quoted_url .. "); " ..

        "$req.Method = 'GET'; " ..
        "$req.UserAgent = 'Mozilla/5.0'; " ..
        "$req.Accept = '*/*'; " ..

        -- Conserver le comportement réseau actuel :
        -- connexion directe, sans découverte automatique du proxy.
        "$req.Proxy = " ..
        "[System.Net.GlobalProxySelection]::GetEmptyWebProxy(); " ..

        -- Demander une réponse compressée et la décompresser automatiquement.
        "$req.AutomaticDecompression = " ..
        "[System.Net.DecompressionMethods]::GZip -bor " ..
        "[System.Net.DecompressionMethods]::Deflate; " ..

        "$http = [Diagnostics.Stopwatch]::StartNew(); " ..
        "$resp = $req.GetResponse(); " ..

        "$stream = $resp.GetResponseStream(); " ..

        "$reader = New-Object -TypeName System.IO.StreamReader " ..
        "-ArgumentList @($stream, [System.Text.Encoding]::UTF8); " ..

        "$body = $reader.ReadToEnd(); " ..

        "$reader.Close(); " ..
        "$resp.Close(); " ..

        "$http.Stop(); " ..
        "$total.Stop(); " ..

        "$inv = [Globalization.CultureInfo]::InvariantCulture; " ..

        -- Ces mesures passent par stderr pour ne pas modifier le M3U8.
        "[Console]::Error.WriteLine(" ..
        "'YT_HTTP_SECONDS=' + " ..
        "$http.Elapsed.TotalSeconds.ToString('F3', $inv)); " ..

        "[Console]::Error.WriteLine(" ..
        "'YT_PS_SECONDS=' + " ..
        "$total.Elapsed.TotalSeconds.ToString('F3', $inv)); " ..

        "$utf8 = New-Object -TypeName System.Text.UTF8Encoding " ..
        "-ArgumentList @($false); " ..

        "[Console]::OutputEncoding = $utf8; " ..
        "[Console]::Out.Write($body)"

    local subprocess_start = mp.get_time()

    local ok, result = pcall(utils.subprocess, {
        args = {
            'powershell.exe',
            '-NoLogo',
            '-NoProfile',
            '-NonInteractive',
            '-ExecutionPolicy',
            'Bypass',
            '-Command',
            ps
        },
        capture_stdout = true,
        capture_stderr = true,
        max_size = MAX_MANIFEST_SIZE,
        playback_only = true
    })

    local subprocess_seconds = mp.get_time() - subprocess_start

    if not ok then
        return nil,
            'impossible de lancer PowerShell : ' .. tostring(result)
    end

    if not result then
        return nil, 'aucun résultat renvoyé par PowerShell'
    end

    -- Afficher les mesures internes pour distinguer le temps réseau
    -- du coût de lancement et d'exécution de PowerShell.
    local stderr_text =
        type(result.stderr) == 'string' and result.stderr or ''

    local http_seconds =
        tonumber(stderr_text:match('YT_HTTP_SECONDS=([%d%.]+)'))

    local ps_seconds =
        tonumber(stderr_text:match('YT_PS_SECONDS=([%d%.]+)'))

    if http_seconds and ps_seconds then
        mp.msg.info(string.format(
            'yt-hls-select : HTTP=%.2f s ; PowerShell hors HTTP=%.2f s ; subprocess total=%.2f s',
            http_seconds,
            math.max(0, ps_seconds - http_seconds),
            subprocess_seconds))
    else
        mp.msg.info(string.format(
            'yt-hls-select : durée totale du subprocess=%.2f s ; mesure HTTP indisponible',
            subprocess_seconds))
    end

    if result.status ~= 0 then
        local detail =
            result.stderr or result.error or result.error_string
            or 'erreur inconnue'

        return nil,
            'téléchargement du manifeste échoué : ' .. tostring(detail)
    end

    local body = result.stdout

    if type(body) ~= 'string'
        or not body:find('#EXTM3U', 1, true) then
        return nil,
            'la réponse reçue ne ressemble pas à un manifeste M3U8'
    end

    return body
end

-- Lit les entrées audio et les variantes vidéo du manifeste maître.
local function parse_master(manifest)
    local lines = {}

    for line in manifest:gmatch('[^\r\n]+') do
        lines[#lines + 1] = line
    end

    local media = {}
    local streams = {}
    local i = 1

    while i <= #lines do
        local line = trim(lines[i])

        if line:find('#EXT-X-MEDIA:', 1, true) == 1 then
            media[#media + 1] = {
                line = line,
                attrs = parse_hls_attributes(line)
            }

        elseif line:find('#EXT-X-STREAM-INF:', 1, true) == 1 then
            local attrs = parse_hls_attributes(line)
            local j = i + 1

            while j <= #lines and trim(lines[j]) == '' do
                j = j + 1
            end

            local uri = j <= #lines and trim(lines[j]) or nil

            streams[#streams + 1] = {
                line = line,
                attrs = attrs,
                uri = uri
            }

            if uri then
                i = j
            end
        end

        i = i + 1
    end

    return media, streams
end


local function get_stream_height(stream)
    local resolution = stream.attrs.RESOLUTION or ''
    return tonumber(resolution:match('x(%d+)'))
end


local function stream_is_h264(stream)
    local codecs = lower(stream.attrs.CODECS)

    return codecs:find('avc1', 1, true) ~= nil
        or codecs:find('avc3', 1, true) ~= nil
        or codecs:find('h264', 1, true) ~= nil
end


local function choose_stream(streams)
    local candidates = {}

    for _, s in ipairs(streams) do
        local height = get_stream_height(s)

        if height
            and s.uri
            and s.uri:match('^https?://') then

            s.height = height
            s.h264 = stream_is_h264(s)

            candidates[#candidates + 1] = s
        end
    end

    if #candidates == 0 then
        return nil
    end

    local best_height = nil

    for _, s in ipairs(candidates) do
        if s.height <= TARGET_H
            and (best_height == nil or s.height > best_height) then

            best_height = s.height
        end
    end

    if best_height == nil then
        for _, s in ipairs(candidates) do
            if best_height == nil or s.height < best_height then
                best_height = s.height
            end
        end
    end

    local chosen = nil

    for _, s in ipairs(candidates) do
        if s.height == best_height then
            if chosen == nil then
                chosen = s
            end

            if PREFER_H264 and s.h264 then
                chosen = s
                break
            end
        end
    end

    return chosen
end


-- Choisit l'audio original dans le groupe associé à la vidéo.
-- Si aucun original n'est trouvé dans ce groupe, cherche dans les
-- autres groupes, puis utilise DEFAULT=YES ou la première piste.
local function choose_audio(audio_rows, group_id)
    local group_rows = {}

    for _, row in ipairs(audio_rows) do
        if row.attrs['GROUP-ID'] == group_id then
            group_rows[#group_rows + 1] = row
        end
    end

    for _, row in ipairs(group_rows) do
        if contains_original(row.attrs.NAME)
            and row.attrs.URI then
            return row, group_rows
        end
    end

    for _, row in ipairs(audio_rows) do
        if contains_original(row.attrs.NAME)
            and row.attrs.URI then

            local other_group = row.attrs['GROUP-ID']
            local other_rows = {}

            for _, candidate in ipairs(audio_rows) do
                if candidate.attrs['GROUP-ID'] == other_group then
                    other_rows[#other_rows + 1] = candidate
                end
            end

            return row, other_rows
        end
    end

    for _, row in ipairs(group_rows) do
        if lower(row.attrs.DEFAULT) == 'yes'
            and row.attrs.URI then
            return row, group_rows
        end
    end

    for _, row in ipairs(group_rows) do
        if row.attrs.URI then
            return row, group_rows
        end
    end

    return nil, group_rows
end


local function on_load()
    local url =
        mp.get_property('stream-open-filename')
        or mp.get_property('path')
        or ''

    if not lower(url):find(
        'manifest.googlevideo.com/api/manifest/hls_variant', 1, true) then
        return
    end

    pending_audio = nil
    audio_add_started = false

    mp.msg.info('yt-hls-select : analyse précoce du manifeste HLS')

    local start_time = mp.get_time()
    local manifest, fetch_error = fetch_manifest(url)

    if not manifest then
        mp.msg.warn('yt-hls-select : ' .. tostring(fetch_error) ..
            ' ; retour à la lecture normale de MPV')
        return
    end

    local media, streams = parse_master(manifest)
    local selected_stream = choose_stream(streams)

    if not selected_stream then
        mp.msg.warn('yt-hls-select : aucune variante vidéo exploitable')
        return
    end

    local audio_rows = {}

    for _, row in ipairs(media) do
        if lower(row.attrs.TYPE) == 'audio' then
            audio_rows[#audio_rows + 1] = row
        end
    end

    local selected_audio, audio_group_rows =
        choose_audio(audio_rows, selected_stream.attrs.AUDIO)

    if not selected_audio or not selected_audio.attrs.URI
        or not selected_audio.attrs.URI:match('^https?://') then

        mp.msg.warn(
            'yt-hls-select : aucune URL audio externe exploitable ; ' ..
            'retour à la lecture normale de MPV')
        return
    end

    -- Mémoriser les pistes audio du groupe choisi sans les ouvrir.
audio_catalog = {}
loaded_audio_ids = {}
current_audio_track_id = nil
pending_audio_load = nil

for _, row in ipairs(audio_group_rows) do
    local audio_url = row.attrs.URI

    if audio_url and audio_url:match('^https?://') then
        audio_catalog[#audio_catalog + 1] = {
            url = audio_url,
            name = row.attrs.NAME
                or row.attrs.LANGUAGE
                or 'Audio',
            lang = row.attrs.LANGUAGE or ''
        }
    end
end

-- Repérer la position de la piste choisie au démarrage.
local selected_index = nil

for i, entry in ipairs(audio_catalog) do
    if entry.url == selected_audio.attrs.URI then
        selected_index = i
        break
    end
end

-- Secours : ajouter la piste sélectionnée si elle ne figure pas
-- dans le catalogue pour une raison inattendue.
if not selected_index then
    table.insert(audio_catalog, 1, {
        url = selected_audio.attrs.URI,
        name = selected_audio.attrs.NAME or 'Audio original',
        lang = selected_audio.attrs.LANGUAGE or ''
    })

    selected_index = 1
end

current_audio_index = selected_index

-- Au démarrage, seule cette piste sera chargée.
pending_audio = {
    url = selected_audio.attrs.URI,
    name = selected_audio.attrs.NAME or 'Audio original',
    lang = selected_audio.attrs.LANGUAGE or '',
    index = selected_index
}


    -- Point essentiel : on ouvre directement le flux vidéo distant.
    -- On ne redirige plus MPV vers un fichier M3U8 local.
    local set_ok, set_error =
        mp.set_property('stream-open-filename', selected_stream.uri)

    if not set_ok then
        pending_audio = nil
        mp.msg.warn('yt-hls-select : redirection vidéo impossible : ' ..
            tostring(set_error))
        return
    end

    mp.msg.info(string.format(
        'yt-hls-select : vidéo %dp (%s) ; audio choisi : %s (%s) ; manifeste analysé en %.2f s',
        selected_stream.height,
        tostring(selected_stream.attrs.CODECS),
        tostring(selected_audio.attrs.NAME),
        tostring(selected_audio.attrs.LANGUAGE),
        mp.get_time() - start_time))
end


local function add_audio_track(url, flags, title, lang)
    local ok, err = pcall(
        mp.commandv,
        'audio-add',
        url,
        flags,
        title,
        lang
    )

    if not ok then
        mp.msg.error(
            'yt-hls-select : impossible d’ajouter la piste audio "' ..
            tostring(title) .. '" : ' .. tostring(err))
        return false
    end

    return true
end


local function on_preloaded()
    if not pending_audio or audio_add_started then
        return
    end

    local audio = pending_audio

    audio_add_started = true
    pending_audio = nil

    pending_audio_load = {
        url = audio.url,
        name = audio.name,
        lang = audio.lang,
        index = audio.index,
        old_id = nil,
        old_index = nil
    }

    mp.msg.info(
        'yt-hls-select : chargement audio initial : "' ..
        tostring(audio.name) .. '"'
    )

    local added = add_audio_track(
        audio.url,
        'select',
        audio.name,
        audio.lang
    )

    if not added then
        pending_audio_load = nil
        return
    end

    -- Éviter d'attendre indéfiniment si MPV n'ajoute pas la piste.
    mp.add_timeout(15, function()
        if pending_audio_load
            and pending_audio_load.url == audio.url then

            mp.msg.warn(
                'yt-hls-select : délai dépassé lors du chargement audio : ' ..
                tostring(audio.name)
            )

            pending_audio_load = nil
        end
    end)
end

-- Affichage du nom et de la position de la piste.
local function audio_label(index)
    local entry = audio_catalog[index]

    if not entry then
        return 'Piste audio inconnue'
    end

    local label = entry.name

    if entry.lang ~= '' then
        label = label .. ' [' .. entry.lang .. ']'
    end

    return string.format(
        'Audio %d/%d : %s',
        index,
        #audio_catalog,
        label
    )
end


-- Termine le changement dès que MPV signale l'ajout de la piste.
local function on_track_list_changed(_, tracks)
    local pending = pending_audio_load

    if not pending or type(tracks) ~= 'table' then
        return
    end

    for _, track in ipairs(tracks) do
        if track.type == 'audio'
            and track.external == true
            and tostring(track.id) ~= tostring(pending.old_id) then

            local filename = track['external-filename']
            local matches_url = filename == pending.url

            -- Secours si MPV ne fournit pas le nom du fichier externe.
            local matches_title = not filename
                and track.title == pending.name
                and (
                    pending.lang == ''
                    or lower(track.lang) == lower(pending.lang)
                )

            if matches_url or matches_title then
                local new_id = track.id

                pending_audio_load = nil

                local selected_ok, selected_error =
                    mp.set_property('aid', tostring(new_id))

                if not selected_ok then
                    mp.msg.warn(
                        'yt-hls-select : impossible de sélectionner la piste : ' ..
                        tostring(selected_error)
                    )
                    return
                end

                loaded_audio_ids[pending.index] = new_id
                current_audio_index = pending.index
                current_audio_track_id = new_id

                -- Retirer la piste externe précédente après la sélection
                -- de la nouvelle, afin de ne pas accumuler les flux audio.
                if pending.old_id
                    and tostring(pending.old_id) ~= tostring(new_id) then

                    local ok, err = pcall(
                        mp.commandv,
                        'audio-remove',
                        tostring(pending.old_id)
                    )

                    if not ok then
                        mp.msg.warn(
                            'yt-hls-select : retrait de l’ancienne piste impossible : ' ..
                            tostring(err)
                        )
                    end

                    if pending.old_index
                        and tostring(loaded_audio_ids[pending.old_index])
                            == tostring(pending.old_id) then

                        loaded_audio_ids[pending.old_index] = nil
                    end
                end

                mp.osd_message(audio_label(current_audio_index), 3)

                mp.msg.info(
                    'yt-hls-select : piste audio active : ' ..
                    audio_label(current_audio_index)
                )

                return
            end
        end
    end
end

mp.observe_property(
    'track-list',
    'native',
    on_track_list_changed
)


-- Charge une langue uniquement lorsqu'elle est demandée.
local function select_audio_index(index)
    if #audio_catalog == 0 then
        mp.osd_message('Aucune piste audio répertoriée', 3)
        return
    end

    index = tonumber(index)

    if not index then
        return
    end

    index = math.floor(index)

    if index < 1 or index > #audio_catalog then
        mp.osd_message('Numéro de piste audio invalide', 3)
        return
    end

    if pending_audio_load then
        mp.osd_message('Chargement audio déjà en cours', 2)
        return
    end

    if index == current_audio_index and current_audio_track_id then
        mp.osd_message(audio_label(index), 3)
        return
    end

    local entry = audio_catalog[index]
    local old_id = current_audio_track_id
    local old_index = current_audio_index

    -- Si la piste est déjà chargée, la sélectionner directement.
    local loaded_id = loaded_audio_ids[index]

    if loaded_id then
        local selected_ok, selected_error =
            mp.set_property('aid', tostring(loaded_id))

        if not selected_ok then
            mp.msg.warn(
                'yt-hls-select : sélection audio impossible : ' ..
                tostring(selected_error)
            )
            return
        end

        current_audio_index = index
        current_audio_track_id = loaded_id

        if old_id
            and tostring(old_id) ~= tostring(loaded_id) then

            pcall(
                mp.commandv,
                'audio-remove',
                tostring(old_id)
            )

            loaded_audio_ids[old_index] = nil
        end

        mp.osd_message(audio_label(index), 3)
        return
    end

    -- La piste n'est pas encore chargée : ouvrir uniquement son URL.
    pending_audio_load = {
        url = entry.url,
        name = entry.name,
        lang = entry.lang,
        index = index,
        old_id = old_id,
        old_index = old_index
    }

    mp.osd_message(
        'Chargement audio : ' .. entry.name,
        3
    )

    local added = add_audio_track(
        entry.url,
        'select',
        entry.name,
        entry.lang
    )

    if not added then
        pending_audio_load = nil
        return
    end

    mp.add_timeout(15, function()
        if pending_audio_load
            and pending_audio_load.url == entry.url then

            mp.msg.warn(
                'yt-hls-select : délai dépassé pour la piste ' ..
                tostring(entry.name)
            )

            pending_audio_load = nil
        end
    end)
end


local function next_audio()
    if #audio_catalog == 0 then
        return
    end

    local next_index = (current_audio_index % #audio_catalog) + 1
    select_audio_index(next_index)
end


local function previous_audio()
    if #audio_catalog == 0 then
        return
    end

    local previous_index =
        ((current_audio_index - 2) % #audio_catalog) + 1

    select_audio_index(previous_index)
end


local function show_audio_catalog()
    if #audio_catalog == 0 then
        mp.osd_message('Aucune piste audio répertoriée', 3)
        return
    end

    local lines = {
        'LANGUES AUDIO — Alt+A : suivante ; Alt+Maj+A : précédente',
        ''
    }

    for i, entry in ipairs(audio_catalog) do
        local marker = (i == current_audio_index) and '* ' or '  '
        local label = entry.name

        if entry.lang ~= '' then
            label = label .. ' [' .. entry.lang .. ']'
        end

        lines[#lines + 1] = string.format(
            '%s%d. %s',
            marker,
            i,
            label
        )
    end

    mp.osd_message(table.concat(lines, '\n'), 8)
end
